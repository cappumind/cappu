MODELO DE REGRAS DO FIRESTORE — CAPPU MIND

Leia antes de publicar: este arquivo é um MODELO consolidado para as coleções identificadas no código atual. Compare com as regras já existentes no Console do Firebase e preserve regras de outras coleções que seu projeto realmente usa. As regras não devem ser substituídas sem revisão.

Pontos de segurança:
- Um leitor só pode criar o próprio documento Leitores/{uid} com perfil "cliente".
- O leitor não pode alterar o próprio campo perfil para "admin".
- A promoção de outro leitor a administrador só é permitida se quem executa a operação já for administrador.
- O texto completo de rascunhos não fica disponível a visitantes.
- As permissões são verificadas no Firestore, não apenas pela interface do site.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }
    function isSelf(leitorId) {
      return signedIn() && request.auth.uid == leitorId;
    }
    function isAdmin() {
      return signedIn()
        && exists(/databases/$(database)/documents/Leitores/$(request.auth.uid))
        && get(/databases/$(database)/documents/Leitores/$(request.auth.uid)).data.perfil == 'admin';
    }
    function ownsStory(storyId) {
      return signedIn()
        && exists(/databases/$(database)/documents/historias/$(storyId))
        && get(/databases/$(database)/documents/historias/$(storyId)).data.autorUid == request.auth.uid;
    }

    match /Leitores/{leitorId} {
      allow create: if isSelf(leitorId)
        && request.resource.data.perfil == 'cliente'
        && request.resource.data.email is string
        && (!('nome' in request.resource.data) || request.resource.data.nome is string);
      allow read: if isSelf(leitorId) || isAdmin();
      allow update: if isAdmin() || (isSelf(leitorId)
        && request.resource.data.perfil == resource.data.perfil
        && request.resource.data.email == resource.data.email);
      allow delete: if isAdmin();

      match /favoritos/{storyId} {
        allow read, delete: if isSelf(leitorId);
        allow create, update: if isSelf(leitorId)
          && request.resource.data.storyId == storyId
          && request.resource.data.titulo is string;
      }
    }

    match /historias/{storyId} {
      allow read: if isAdmin()
        || resource.data.status == 'publicada'
        || (signedIn() && resource.data.autorUid == request.auth.uid);
      allow create: if signedIn()
        && request.resource.data.autorUid == request.auth.uid
        && request.resource.data.titulo is string
        && request.resource.data.titulo.size() > 0
        && request.resource.data.titulo.size() <= 100
        && request.resource.data.sinopse is string
        && request.resource.data.sinopse.size() <= 500
        && request.resource.data.genero is string
        && (request.resource.data.status in ['rascunho', 'publicada']
          || (isAdmin() && request.resource.data.status == 'bloqueada'));
      allow update: if signedIn()
        && (isAdmin() || resource.data.autorUid == request.auth.uid)
        && request.resource.data.autorUid == resource.data.autorUid
        && request.resource.data.titulo is string
        && request.resource.data.titulo.size() > 0
        && request.resource.data.titulo.size() <= 100
        && request.resource.data.sinopse is string
        && request.resource.data.sinopse.size() <= 500
        && request.resource.data.genero is string
        && request.resource.data.status in ['rascunho', 'publicada', 'bloqueada']
        && (isAdmin() || request.resource.data.status in ['rascunho', 'publicada']);
      allow delete: if isAdmin() || (signedIn() && resource.data.autorUid == request.auth.uid);

      match /conteudo/{contentId} {
        allow read: if signedIn()
          && (isAdmin()
            || get(/databases/$(database)/documents/historias/$(storyId)).data.status == 'publicada'
            || ownsStory(storyId));
        allow create, update: if signedIn()
          && (isAdmin() || ownsStory(storyId))
          && contentId == 'full'
          && request.resource.data.texto is string
          && request.resource.data.texto.size() <= 200000;
        allow delete: if isAdmin() || (signedIn() && ownsStory(storyId));
      }
    }

    match /comunidade/{postId} {
      allow read: if true;
      allow create: if signedIn()
        && request.resource.data.autorUid == request.auth.uid
        && request.resource.data.texto is string
        && request.resource.data.texto.size() > 0
        && request.resource.data.texto.size() <= 1200
        && request.resource.data.tipo is string
        && request.resource.data.autorNome is string;
      allow update: if signedIn()
        && resource.data.autorUid == request.auth.uid
        && request.resource.data.autorUid == resource.data.autorUid
        && request.resource.data.texto is string
        && request.resource.data.texto.size() > 0
        && request.resource.data.texto.size() <= 1200;
      allow delete: if signedIn() && resource.data.autorUid == request.auth.uid;
    }

    // Coleção antiga mantida para compatibilidade com o painel administrativo atual.
    match /produtos/{produtoId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }
    match /compras/{compraId} {
      allow create: if signedIn() && request.resource.data.uid == request.auth.uid;
      allow read: if signedIn() && (resource.data.uid == request.auth.uid || isAdmin());
      allow update, delete: if isAdmin();
    }

    // A coleção antiga "usuarios" só deve permanecer se alguma página ainda a utilizar.
    match /usuarios/{uid} {
      allow read: if isSelf(uid) || isAdmin();
      allow create: if isSelf(uid) && request.resource.data.perfil != 'admin';
      allow update: if isAdmin() || (isSelf(uid)
        && request.resource.data.perfil == resource.data.perfil);
      allow delete: if isAdmin();
    }

    // Sem permissões genéricas: coleções não listadas ficam negadas por padrão.
  }
}
