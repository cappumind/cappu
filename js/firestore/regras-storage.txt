MODELO DE REGRAS DO FIREBASE STORAGE — CAPPU MIND

Use somente se a página administrativa continuar enviando imagens para Storage. Revise com as regras atuais antes de publicar.

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    function signedIn() { return request.auth != null; }
    function isAdmin() {
      return signedIn()
        && firestore.exists(/databases/(default)/documents/Leitores/$(request.auth.uid))
        && firestore.get(/databases/(default)/documents/Leitores/$(request.auth.uid)).data.perfil == 'admin';
    }
    match /produtos/{fileName} {
      allow read: if true;
      allow write: if isAdmin()
        && request.resource.size < 5 * 1024 * 1024
        && request.resource.contentType.matches('image/.*');
    }
  }
}
